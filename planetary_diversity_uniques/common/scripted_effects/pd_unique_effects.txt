#Floating World
pd_create_artifical_floating_islands_effect = {
	if = {
		limit = {
			is_wet = yes
		}
		create_ambient_object = {
			type = pd_floating_islands_wet_object
			location = this
			use_3d_location = yes
			entity_scale_to_size = yes
		}
		last_created_ambient_object = {
			set_ambient_object_flag = pd_floating_islands
		}
	}
	if = {
		limit = {
			is_dry = yes
		}
		create_ambient_object = {
			type = pd_floating_islands_dry_object
			location = this
			use_3d_location = yes
			entity_scale_to_size = yes
		}
		last_created_ambient_object = {
			set_ambient_object_flag = pd_floating_islands
		}
	}
	if = {
		limit = {
			is_cold = yes
		}
		create_ambient_object = {
			type = pd_floating_islands_cold_object
			location = this
			use_3d_location = yes
			entity_scale_to_size = yes
		}
		last_created_ambient_object = {
			set_ambient_object_flag = pd_floating_islands
		}
	}
}

#Bonefield World
pd_create_bonefield_tiyanki_country_effect = {
	if = {
		limit = {
			NOT = { exists = event_target:pd_bonefield_tiyanki_country }
		}
		create_country = {
			name = "NAME_Placid_Leviathans"
			type = tiyanki_garrison
			flag = {
				icon = {
					category = "zoological"
					file = "flag_zoological_1.dds"
				}
				background = {
					category = "backgrounds"
					file = "00_solid.dds"
				}
				colors = { "black" "black" "null" "null" }
			}
			effect = {
				save_global_event_target_as = pd_bonefield_tiyanki_country
			}
		}
	}
}

# Planet scope. Fall boost on a Controlled Bonefield, refreshed by each fall; pdbonefield.110 applies it, pdbonefield.120 previews it.
pd_bonefield_controlled_fall_payout_effect = {
	add_modifier = {
		modifier = pd_bonefield_fresh_carcass
		days = 3650
	}
}

# Planet scope.
pd_bonefield_add_controlled_effect = {
	if = {
		limit = {
			owner = { is_gestalt = yes }
		}
		add_modifier = {
			modifier = pd_bonefield_controlled_gestalt
			days = -1
		}
	}
	else = {
		add_modifier = {
			modifier = pd_bonefield_controlled
			days = -1
		}
	}
}

# Planet scope.
pd_bonefield_remove_controlled_effect = {
	remove_modifier = pd_bonefield_controlled
	remove_modifier = pd_bonefield_controlled_gestalt
}
