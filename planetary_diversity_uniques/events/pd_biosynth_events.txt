#Situation Event = Synth Study
situation_event = {
	id = pdbiosynth.210
	title = pdbiosynth.210.name
	desc = pdbiosynth.210.desc
	picture = GFX_evt_pd_biosynth
	location = from
	show_sound = event_mystic_reveal
	fire_only_once = yes

	is_triggered_only = yes

	option = {
		name = pdbiosynth.210.a
		owner = {
			add_monthly_resource_mult = {
				resource = engineering_research
				value = @tier5unityreward
				min = @tier5unitymin
				max = @tier5unitymax
			}
			add_monthly_resource_mult = {
				resource = society_research
				value = @tier5unityreward
				min = @tier5unitymin
				max = @tier5unitymax
			}
		}
		hidden_effect = {
			owner = {
				set_country_flag = unit_04_likes_you
			}
		}
	}

	option = {
		name = pdbiosynth.210.b
		owner = {
			if = {
				limit = {
					has_country_flag = synthetic_empire
				}
				add_monthly_resource_mult = {
					resource = unity
					value = @tier5unityreward
					min = @tier5unitymin
					max = @tier5unitymax
				}
			}
			else_if = {
				limit = { has_technology = tech_synthetic_leaders }
				add_monthly_resource_mult = {
					resource = engineering_research
					value = @tier5researchreward
					min = @tier5researchmin
					max = @tier5researchmax
				}
			}
			else = {
				if = {
					limit = {
						has_technology = tech_sapient_ai
						has_technology = tech_droid_workers
					}
					add_research_option = tech_synthetic_leaders
				}
				else = {
					switch = {
						trigger = has_technology
						tech_sapient_ai = {
							add_monthly_resource_mult = {
								resource = engineering_research
								value = @tier3researchreward
								min = @tier3researchmin
								max = @tier3researchmax
							}
						}
						tech_self_aware_logic = { add_research_option = tech_sapient_ai }
						tech_administrative_ai = { add_research_option = tech_self_aware_logic }
						tech_basic_science_lab_1 = { add_research_option = tech_administrative_ai }
						tech_basic_science_lab_2 = { add_research_option = tech_administrative_ai }
						tech_basic_science_lab_3 = { add_research_option = tech_administrative_ai }
					}
					switch = {
						trigger = has_technology
						tech_droid_workers = {
							add_monthly_resource_mult = {
								resource = engineering_research
								value = @tier3researchreward
								min = @tier3researchmin
								max = @tier3researchmax
							}
						}
						tech_robotic_workers = { add_research_option = tech_droid_workers }
						tech_powered_exoskeletons = { add_research_option = tech_robotic_workers }
						tech_colonial_centralization = { add_research_option = tech_powered_exoskeletons }
					}
				}
			}
		}
		hidden_effect = {
			owner = {
				set_country_flag = unit_04_does_not_like_you
			}
		}
	}
}

# origin_pd_biosynth start: Unit 04 joins the empire as an immortal leader and governs the homeworld. The origin quest later offers to install it permanently.
country_event = {
	id = pdbiosynth.400
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_pd_biosynth
	}

	immediate = {
		create_species = {
			name = NAME_Unit_04
			class = AI
			portrait = unit_04
			immortal = yes
			gender = indeterminate
			traits = { trait = "trait_enduring" }
			effect = {
				save_event_target_as = unit_04_species
			}
		}
		create_leader = {
			class = official
			species = event_target:unit_04_species
			name = NAME_Unit_04
			gender = indeterminable
			immortal = yes
			skill = 4
			custom_description = unit_04_desc
			custom_catch_phrase = unit_04_catch_phrase
			randomize_traits = no
			traits = {
				1 = leader_trait_environmental_engineer
				2 = subclass_official_governor
				3 = leader_trait_homesteader_2
				4 = leader_trait_destructive
			}
			effect = {
				add_trait = { trait = leader_trait_unit_04 }
				set_leader_flag = pd_unit_04
				save_global_event_target_as = pd_unit_04_leader
			}
		}
		capital_scope = {
			assign_leader = event_target:pd_unit_04_leader
			add_building = building_pd_biosynth_assembler
			add_deposit = d_pd_underground_cauldrons
		}
		set_country_flag = pd_has_unit_04
	}
}

# Monthly checker for the Long Watch. Adds one stage of progress per milestone reached, in order: Living Metal, Gene Tailoring, Designed Perfection, then finishing the Biosynthetics tree.
situation_event = {
	id = pdbiosynth.415
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_situation_flag = pd_lw_living_metal }
				owner = { has_technology = tech_mine_living_metal }
			}
			set_situation_flag = pd_lw_living_metal
			add_situation_progress = 100
		}
		else_if = {
