# Overrides vanilla biologist_is_archaeo_engineer_trigger. An ex-relic arcology with the Archeotech Preserve zone swaps Biologists to Archaeo-Engineers without building_archaeostudies_faculty.
biologist_is_archaeo_engineer_trigger = {
	OR = {
		has_active_building = building_archaeostudies_faculty
		num_zones = {
			type = zone_pd_archeotech_preserve
			value > 0
		}
	}
}
