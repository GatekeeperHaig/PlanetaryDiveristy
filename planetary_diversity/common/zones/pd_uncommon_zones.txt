# Uncommon-world specialization zones. Same piggyback pattern as the rare-world set, but gated on the planet-class scripted trigger instead of a modifier, since uncommon worlds are real pc_pd_* classes.
# Rural zones also carry the hive and nexus zone sets: the hive/machine variants of these classes use hive_world/machine_world district sets, whose rural slots (slot_hive / slot_nexus) accept those sets instead of energy/minerals/food. Urban zones need no extra set because district_hive and district_nexus both provide slot_city_01/02.

# ===== Tidally Locked =====
zone_pd_tidally_grid = {
	icon = GFX_district_specialization_energy
	base_buildtime = @zone_buildtime
	potential = {
		hidden_trigger = { exists = owner  owner = { is_wilderness_empire = no } }
		NOT = { has_modifier = resort_colony }
		pd_is_planet_class_tidally_locked = yes
		num_zones = { type = zone_pd_tidally_farm value = 0 }
		num_zones = { type = zone_pd_tidally_lab value = 0 }
	}
	inline_script = "zones/pd_rare_zone_common"
	zone_sets = { energy  hive  nexus }
	included_building_sets = { generator  fallen_empire_generator  generator_automation }
	convert_to = { zone_energy  zone_energy_nexus  zone_energy_hive }
	triggered_district_planet_modifier = { potential = { exists = owner  owner = { is_gestalt = no } }  job_pd_tidally_dayside_collector_add = @scaling_district_1_job }
	triggered_district_planet_modifier = { potential = { exists = owner  owner = { is_gestalt = yes } }  job_pd_tidally_dayside_collector_drone_add = @scaling_district_1_job }
	planet_modifier = { zone_building_slots_add = 3 }
	ai_priority = 5
}
