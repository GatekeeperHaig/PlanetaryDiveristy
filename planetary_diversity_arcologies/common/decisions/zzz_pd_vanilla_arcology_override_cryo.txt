# Overrides the vanilla Arcology Project decision so ammonia/methane/nitrogen worlds become cryo arcologies and the PD capital art re-syncs.
# The pd_is_*_planet triggers stub to always = no here and are overridden by PD Exotics, so the cryo branch only fires when Exotics is loaded.

##########################################################################
# Vanilla: Arcology Project
##########################################################################
decision_arcology_project = {
	owned_planets_only = yes
	icon = decision_arcology_project

	enactment_time = 3600
	resources = {
		category = decisions
		cost = {
			influence = 200
			alloys = 10000
		}
	}

	potential = {
		NOR = {
			is_planet_class = pc_city
			is_planet_class = pc_relic
			is_planet_class = pc_nanotech
			has_modifier = resort_colony
			has_modifier = penal_colony
			has_modifier = slave_colony
			has_modifier = pm_wenkwort_custodian
			has_carrier_flag = resort_colony_queued
		}
		is_artificial = no
		exists = owner
		owner = { has_ascension_perk = ap_arcology_project }
		NOT = { has_designation = col_nanotech_world }
	}

	allow = {
		if = {
			limit = { owner = { is_ai = no } }
			custom_tooltip = {
				fail_text = decision_arcology_project_requirements
				free_district_slots < 1
				num_districts = { type = district_farming value < 1 }
				num_districts = { type = district_mining value < 1 }
				num_districts = { type = district_generator value < 1 }
				num_districts = { type = district_geothermal value < 1 }
				num_districts = { type = district_melting value < 1 }
				num_districts = { type = district_polytechnic value < 1 }
			}
			custom_tooltip = {
				fail_text = legendary_leader_decision_arcology_project
				NOT = { has_planet_flag = legendary_leader_planet }
			}
		}
		num_uncleared_blockers < 1
	}

	on_queued = {
		set_carrier_flag = arcology_project_construction
	}

	on_unqueued = {
		remove_carrier_flag = arcology_project_construction
	}

	effect = {
		remove_carrier_flag = arcology_project_construction
		custom_tooltip = decision_arcology_project_effects
		if = {
			limit = { pd_is_ammonia_planet = yes }
			pd_add_ammonia_atmosphere_effect = yes
		}
		else_if = {
			limit = { pd_is_methane_planet = yes }
			pd_add_methane_atmosphere_effect = yes
		}
		else_if = {
			limit = { pd_is_nitrogen_planet = yes }
			pd_add_nitrogen_atmosphere_effect = yes
		}
		hidden_effect = {
			if = {
				limit = {
					OR = {
						pd_has_ammonia_atmosphere = yes
						pd_has_methane_atmosphere = yes
						pd_has_nitrogen_atmosphere = yes
					}
				}
				pd_exotic_cryo_arcology_convert_effect = yes
			}
			else = {
				save_deposits = yes
				clear_deposits = yes
				restore_saved_deposits = yes
				change_pc = pc_city
				relic_world_conversion_effect = {
					DISTRICT = district_arcology_housing
					VARIABLE = num_housing_districts
				}
				set_planet_entity = {
					entity = "city_planet_entity"
					graphical_culture = owner
				}
				if = {
					limit = { has_pd_capital_deposit = yes }
					pd_resync_capital_art_effect = yes
				}
				carrier_event = { id = action.68 }
				owner = {
					country_event = { id = distar.201 }
				}
				carrier_event = { id = mega.200 }
				if = {
					limit = {
						OR = {
							is_active_resolution = "resolution_ecology_environmental_control_board"
							is_active_resolution = "resolution_ecology_paradise_initiative"
						}
					}
					owner = {
						set_timed_country_flag = {
							flag = resolution_breached_terraformed
							days = 3600
						}
					}
				}
			}
		}
	}

	ai_weight = {
		weight = 1
		modifier = {
			factor = 100
			exists = owner
			OR = {
				pop_amount >= 7000
				is_capital = yes
			}
		}
	}
}
