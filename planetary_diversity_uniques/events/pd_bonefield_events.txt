	id = pdbonefield.10

		from = {

		}

			NOT = {
				has_country_flag = pd_bonefield_encounter
			}
		}
	}

		owner = {
			set_country_flag = pd_bonefield_encounter
		}
	}

	option = {
		name = pdbonefield.10.a
	}
}

#Pilgrimage spawn. A dying tiyanki enters at the system edge and is ordered to the Bonefield; arriving fires pdbonefield.110.
country_event = {
	id = pdbonefield.100
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		any_owned_planet = {
			planet = {
				has_planet_flag = pd_bonefield_planet
				NOT = { has_modifier = pd_bonefield_warded }
			}
		}
	}

	immediate = {
		random_list = {
			60 = {
				pd_create_bonefield_tiyanki_country_effect = yes
				random_owned_planet = {
					limit = {
						planet = {
							has_planet_flag = pd_bonefield_planet
							NOT = { has_modifier = pd_bonefield_warded }
						}
					}
					solar_system = {
						random_system_planet = {
							limit = { is_star = yes }
							save_event_target_as = pd_bonefield_star
						}
					}
					event_target:pd_bonefield_tiyanki_country = {
						create_fleet = {
							name = "NAME_Tiyanki_Space_Whale"
							settings = { spawn_debris = no }
							effect = {
								set_owner = event_target:pd_bonefield_tiyanki_country
								create_ship = { design = "NAME_Tiyanki_Cow" }
								create_ship = { design = "NAME_Tiyanki_Bull" }
								set_location = {
									target = event_target:pd_bonefield_star
									distance = 260
									angle = random
								}
								set_fleet_stance = passive
								set_aggro_range_measure_from = return_point
								set_aggro_range = 0
								set_fleet_flag = NAME_Tiyanki_Space_Whale
								set_fleet_flag = pd_bonefield_pilgrim
								queue_actions = {
									find_closest_planet = {
										trigger = {
											id = pd.bonefield.pilgrimage
											has_carrier_flag = pd_bonefield_planet
										}
										found_planet = {
											move_to = this
											effect = {
												id = pd.bonefield.pilgrimage.arrival
												solar_system = {
													every_system_planet = {
														limit = { has_planet_flag = pd_bonefield_planet }
														planet_event = { id = pdbonefield.110 }
													}
												}
											}
										}
									}
								}
							}
						}
					}
				}
			}
			40 = { }
		}
	}
}

#Pilgrimage resolution, fired on arrival. Despawns the fleet; base and Controlled Bonefields get their timed fall modifier, Warded gets nothing.
planet_event = {
	id = pdbonefield.110
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		exists = owner
		has_planet_flag = pd_bonefield_planet
	}

	immediate = {
		solar_system = {
			every_fleet_in_system = {
				limit = {
					has_fleet_flag = pd_bonefield_pilgrim
				}
				destroy_fleet = this
			}
		}
		if = {
			limit = {
				NOT = { has_modifier = pd_bonefield_warded }
			}
			if = {
				limit = {
					pd_has_bonefield_controlled = yes
				}
				pd_bonefield_controlled_fall_payout_effect = yes
			}
			if = {
				limit = {
					has_modifier = pd_bonefield
				}
				add_modifier = {
					modifier = pd_bonefield_wild_fall
					days = 3650
				}
			}
			owner = {
				create_message = {
					type = MESSAGE_PD_TIYANKI_FALL
					localization = MESSAGE_PD_TIYANKI_FALL_DESC
					days = 30
					target = root
					variable = {
						type = name
						localization = PLANET
						scope = root
					}
				}
			}
		}
	}
}

#End of stage 1. Sets the outcome flag read by pdbonefield.130.
situation_event = {
	id = pdbonefield.120
	title = pdbonefield.120.name
	desc = pdbonefield.120.desc
	picture = GFX_evt_pd_bonefield_surface
	location = from
	show_sound = event_mystic_reveal
	fire_only_once = yes

	is_triggered_only = yes

	option = {
		name = pdbonefield.120.a
		custom_tooltip = pdbonefield.120.a.tooltip
		tooltip = {
			target = {
				pd_bonefield_add_controlled_effect = yes
			}
		}
		custom_tooltip = pdbonefield.120.fall_header
		tooltip = {
			target = {
				pd_bonefield_controlled_fall_payout_effect = yes
			}
		}
		set_situation_flag = pd_bonefield_path_control
	}

	option = {
		name = pdbonefield.120.b
		custom_tooltip = pdbonefield.120.b.tooltip
		tooltip = {
			target = {
				add_modifier = {
					modifier = pd_bonefield_warded
					days = -1
				}
			}
		}
		custom_tooltip = pdbonefield.120.fall_header
		custom_tooltip = pdbonefield.120.b.falls_tt
		set_situation_flag = pd_bonefield_path_stop
	}
}

#Situation completion. Applies the outcome modifier for the path chosen in pdbonefield.120.
situation_event = {
	id = pdbonefield.130
	title = pdbonefield.130.name
	location = from
	show_sound = event_mystic_reveal
	fire_only_once = yes

	is_triggered_only = yes

	picture = {
		trigger = {
			NOT = { has_situation_flag = pd_bonefield_path_stop }
		}
		picture = GFX_evt_tiyanki_family
	}

	picture = {
		trigger = {
			has_situation_flag = pd_bonefield_path_stop
		}
		picture = GFX_evt_pd_bonefield_surface
	}

	desc = {
		text = pdbonefield.130.desc.control
		trigger = {
			NOT = { has_situation_flag = pd_bonefield_path_stop }
		}
	}

	desc = {
		text = pdbonefield.130.desc.stop
		trigger = {
			has_situation_flag = pd_bonefield_path_stop
		}
	}

	option = {
		name = pdbonefield.130.a
		trigger = {
			NOT = { has_situation_flag = pd_bonefield_path_stop }
		}
		target = {
			remove_modifier = pd_bonefield
			remove_modifier = pd_bonefield_wild_fall
			pd_bonefield_add_controlled_effect = yes
			set_planet_flag = pd_bonefield_choice_made
		}
		destroy_situation = this
	}

	option = {
		name = pdbonefield.130.b
		trigger = {
			has_situation_flag = pd_bonefield_path_stop
		}
		target = {
			remove_modifier = pd_bonefield
			remove_modifier = pd_bonefield_wild_fall
			add_modifier = {
				modifier = pd_bonefield_warded
				days = -1
			}
			set_planet_flag = pd_bonefield_choice_made
		}
		destroy_situation = this
	}
}

#Owner change. Swaps between the regular and gestalt Controlled modifiers to match the new owner.
carrier_event = {
	id = pdbonefield.140
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_planet_flag = pd_bonefield_planet
		exists = owner
		pd_has_bonefield_controlled = yes
	}

	immediate = {
		pd_bonefield_remove_controlled_effect = yes
		pd_bonefield_add_controlled_effect = yes
	}
}
