# Chemovat Garden rural-district slot for cryogenic planets.

slot_pdexotic_cryo_chemovat = {
	included_zone_sets = {
		pdexotic_cryo_chemovat
	}
	potential = {
		owner? = { is_megacity_architects_empire = no }
	}
	unlock = {
		custom_tooltip = {
			fail_text = zone_unavailable_tech_food_processing_1
			exists = owner
			owner = {
				OR = {
					has_valid_civic = civic_agrarian_idyll
					has_tech_or_functional_civic = {
						TECH = tech_food_processing_1
					}
				}
			}
		}
	}
}
