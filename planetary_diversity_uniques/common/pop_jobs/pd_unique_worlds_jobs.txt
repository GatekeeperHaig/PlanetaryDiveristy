#Biosynth Jobs
pd_biosynth_engineer = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = building_pd_biosynth_cauldron
		}
	}

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		complex_specialist_job_check_trigger = yes
	}

	triggered_planet_modifier = {
		potential = {
			planet = { has_building = building_robot_assembly_plant }
		}
		planet_pop_assembly_add = 0.5
	}

	resources = {
		category = planet_researchers
		produces = {
			engineering_research = 4
			society_research = 4
			sr_living_metal = 0.5
		}
		produces = {
			trigger = {
				planet = {
					has_modifier = pd_biosynth
				}
			}
			engineering_research = -2
			society_research = -2
			sr_living_metal = 1.0
		}
		produces = {
			trigger = {
				planet = { has_building = building_robot_assembly_plant }
			}
			engineering_research = -2
			society_research = -2
		}
		upkeep = {
			consumer_goods = 1.5
			food = 10
		}
	}

	weight = {
		weight = @specialist_job_weight
		modifier = {
			factor = 0.2
			has_citizenship_rights = no
			NOT = { has_trait = trait_mechanical }
		}
		modifier = {
			factor = 2
			has_living_standard = { type = living_standard_academic_privilege }
		}
		modifier = {
			factor = 0.1
			can_take_servant_job = yes
		}
		modifier = {
			factor = 0.8
			is_enslaved = yes
			has_slavery_type = { type = slavery_indentured }
		}
		modifier = {
			factor = 5
			has_trait = trait_cyborg_creed_of_research
		}
	}

	inline_script = "jobs/automodding_priority_research"

}

# Grows new Biosynthetic pops from living metal. Growth and upkeep scale per worker, and both halt on a living metal deficit.
pd_biosynth_cultivator = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = building_pd_biosynth_cauldron
		}
	}

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		complex_specialist_job_check_trigger = yes
	}

	triggered_planet_pop_group_modifier_for_species = {
		potential = {
			has_trait = trait_biosynth
			owner = { NOT = { has_deficit = sr_living_metal } }
		}
		bonus_pop_growth = 4
	}
	triggered_planet_pop_group_modifier_for_species = {
		potential = {
			has_trait = trait_biosynth
			owner = {
				NOT = { has_deficit = sr_living_metal }
				has_tradition = pd_tr_biosynth_perk_2a
			}
		}
		bonus_pop_growth = 2
	}

	resources = {
		category = planet_pop_assemblers
		upkeep = {
			trigger = { owner = { NOT = { has_tradition = pd_tr_biosynth_perk_2b } } }
			sr_living_metal = 2
		}
		upkeep = {
			trigger = {
				owner = {
					has_tradition = pd_tr_biosynth_perk_2b
					NOT = { has_tradition = pd_tr_biosynth_adopt_finish }
				}
			}
			sr_living_metal = 1.5
		}
		upkeep = {
			trigger = { owner = { has_tradition = pd_tr_biosynth_adopt_finish } }
			sr_living_metal = 1
		}
	}

	weight = {
		weight = @specialist_job_weight
	}
}
