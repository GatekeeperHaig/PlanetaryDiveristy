namespace = pd_exotic

# Populates an exotic homeworld with its class deposits and the planet-class label modifier.
country_event = {
	id = pd_exotic.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = planetarydiversityexotics
		if = {
			limit = {
				exists = capital_scope
				capital_scope = {
					OR = {
						is_planet_class = pc_pd_methane
						is_planet_class = pc_pd_ammonia
						is_planet_class = pc_pd_nitrogen
					}
				}
			}
			capital_scope = {
				generate_pd_exotic_start_deposits = yes

				# Starting rural districts (3 energy / 2 mining / 4 farming, matching a default-origin biological start); the vanilla game-start placement does not deliver these on pdexotic_cryo capitals.
				add_district_and_planet_size_if_needed_effect_amount = {
					amount = 3
					district = district_pdexotic_cryo_riftvent
				}
				add_district_and_planet_size_if_needed_effect_amount = {
					amount = 2
					district = district_pdexotic_cryo_sublimation
				}
				add_district_and_planet_size_if_needed_effect_amount = {
					amount = 4
					district = district_pdexotic_cryo_chemovat
				}

				if = {
					limit = { is_planet_class = pc_pd_ammonia }
					add_modifier = {
						modifier = pd_ammonia
						days = -1
					}
				}
				if = {
					limit = { is_planet_class = pc_pd_methane }
					add_modifier = {
						modifier = pd_methane
						days = -1
					}
				}
				if = {
					limit = { is_planet_class = pc_pd_nitrogen }
					add_modifier = {
						modifier = pd_nitrogen
						days = -1
					}
				}
			}
		}
	}
}

# Labels every naturally-spawning ammonia/methane world at game start (mirrors pdengine.11). Capitals are skipped; pd_exotic.1 handles homeworlds.
country_event = {
	id = pd_exotic.2
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		every_galaxy_planet = {
			limit = {
				is_planet_class = pc_pd_ammonia
				is_capital = no
				NOT = { has_modifier = pd_ammonia }
			}
			add_modifier = { modifier = pd_ammonia days = -1 }
		}
		every_galaxy_planet = {
			limit = {
				is_planet_class = pc_pd_methane
				is_capital = no
				NOT = { has_modifier = pd_methane }
			}
			add_modifier = { modifier = pd_methane days = -1 }
		}
		every_galaxy_planet = {
			limit = {
				is_planet_class = pc_pd_nitrogen
				is_capital = no
				NOT = { has_modifier = pd_nitrogen }
			}
			add_modifier = { modifier = pd_nitrogen days = -1 }
		}
	}
}

# Keeps the ammonia/methane label in sync on planet-class change (terraforming, story events): adds it on entry, strips it on exit.
planet_event = {
	id = pd_exotic.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				is_planet_class = pc_pd_ammonia
				NOT = { has_modifier = pd_ammonia }
			}
			add_modifier = { modifier = pd_ammonia days = -1 }
		}
		else_if = {
			limit = {
				NOT = { is_planet_class = pc_pd_ammonia }
				has_modifier = pd_ammonia
			}
			remove_modifier = pd_ammonia
		}
		if = {
			limit = {
				is_planet_class = pc_pd_methane
				NOT = { has_modifier = pd_methane }
			}
			add_modifier = { modifier = pd_methane days = -1 }
		}
		else_if = {
			limit = {
				NOT = { is_planet_class = pc_pd_methane }
				has_modifier = pd_methane
			}
			remove_modifier = pd_methane
		}
		if = {
			limit = {
				is_planet_class = pc_pd_nitrogen
				NOT = { has_modifier = pd_nitrogen }
			}
			add_modifier = { modifier = pd_nitrogen days = -1 }
		}
		else_if = {
			limit = {
				NOT = { is_planet_class = pc_pd_nitrogen }
				has_modifier = pd_nitrogen
			}
			remove_modifier = pd_nitrogen
		}
	}
}

# on_colonized hook for ap_pd_cryothermal_engineering: if the new colony is cryo and its owner has the AP, stamp the engineered-planet modifier.
carrier_event = {
	id = pd_exotic.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = owner
		owner = { has_ascension_perk = ap_pd_cryothermal_engineering }
		OR = {
			is_planet_class = pc_pd_ammonia
			is_planet_class = pc_pd_methane
			is_planet_class = pc_pd_nitrogen
		}
		NOT = { has_modifier = pd_cryothermal_engineered_planet }
	}

	immediate = {
		add_modifier = { modifier = pd_cryothermal_engineered_planet days = -1 }
	}
}

# on_planet_class_changed hook for ap_pd_cryothermal_engineering: add the modifier when a planet is terraformed into cryo by an AP-empire, remove it when terraformed away.
planet_event = {
	id = pd_exotic.11
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				exists = owner
				owner = { has_ascension_perk = ap_pd_cryothermal_engineering }

					is_planet_class = pc_pd_ammonia
					is_planet_class = pc_pd_methane
					is_planet_class = pc_pd_nitrogen
				}
				NOT = { has_modifier = pd_cryothermal_engineered_planet }
			}
			add_modifier = { modifier = pd_cryothermal_engineered_planet days = -1 }
		}
		else_if = {
			limit = {
				has_modifier = pd_cryothermal_engineered_planet
				NOR = {
					is_planet_class = pc_pd_ammonia
					is_planet_class = pc_pd_methane
					is_planet_class = pc_pd_nitrogen
				}
			}
			remove_modifier = pd_cryothermal_engineered_planet
		}
	}
}

# on_planet_transfer hook for ap_pd_cryothermal_engineering: stamp the modifier if a cryo planet was just transferred to an AP-empire. From scope is the new owner.

	}

	immediate = {
		add_modifier = { modifier = pd_cryothermal_engineered_planet days = -1 }
	}
}

# Stamps pd_ammonia_atmosphere on a new habitat when the builder's main species is ammonia-preference, before colonization (else habitability is 0 at settle time).
# Can't filter on FROM = habitat_central_complex: its on_build_complete removes the megastructure before this fires, so FROM is invalid. Filter on FROMFROM (system) for uncolonized, unmarked habitats.
# Methane mirror is pd_exotic.22.
country_event = {
	id = pd_exotic.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		species = { has_trait = trait_pc_pd_ammonia_preference }
		FROMFROM = {
			any_system_planet = {
				is_planet_class = pc_habitat
				is_colony = no
				NOT = { has_modifier = pd_ammonia_atmosphere }
			}
		}
	}

	immediate = {
		FROMFROM = {
			every_system_planet = {
				limit = {
					is_planet_class = pc_habitat
					is_colony = no
					NOT = { has_modifier = pd_ammonia_atmosphere }
				}
				add_modifier = { modifier = pd_ammonia_atmosphere days = -1 }
				set_planet_entity = {
					picture = pc_pd_ammonia_habitat
					graphical_culture = root
				}
			}
		}
	}
}

# Backstop for habitats that slipped past pd_exotic.20 (restored ruins, mid-build cases): stamps the marker on first colonization by an ammonia-preference empire. Conquest is excluded; a captured habitat keeps its atmosphere until re-colonized.
carrier_event = {
	id = pd_exotic.21
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_planet_class = pc_habitat
		exists = owner
		owner = {
			species = { has_trait = trait_pc_pd_ammonia_preference }
		}
		NOT = { has_modifier = pd_ammonia_atmosphere }
	}

	immediate = {
		add_modifier = { modifier = pd_ammonia_atmosphere days = -1 }
		set_planet_entity = {
			picture = pc_pd_ammonia_habitat
			graphical_culture = owner
		}
	}
}
