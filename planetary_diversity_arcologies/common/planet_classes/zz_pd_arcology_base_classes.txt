# The arcologies submod owns pc_city so its arcology flavors can carry a distinct UI icon via conditional_icon.
# A PD arcology is pc_city + a pd_*_arcology modifier + an entity swap; the conditional_icon swaps the small
# type-icon to match the flavor, falling back to the vanilla city icon for a plain ecumenopolis.
# Body below is VERBATIM 4.4 vanilla pc_city (00_planet_classes_megacorp.txt); re-sync it each patch.
# (All These Worlds also overrides pc_city, but it is out of scope and intentionally ignored.)
pc_city = {
	entity = "city_planet"
	entity_scale = @planet_standard_scale
	icon = GFX_planet_type_city
	icon_large = GFX_planet_type_city_big

	climate = "dry"

	auto_trait_prio = {
		trait_auto_pc_city_preference
	}

	atmosphere_color		= hsv { 0.75 0.10 0.60 }		#DONE
	atmosphere_intensity	= 1.0
	atmosphere_width		= 0.5

	city_color_lut = "gfx/portraits/misc/colorcorrection_continental.dds"

	modifier = {
		planet_jobs_produces_mult = 0.20
		planet_pop_assembly_mult = 0.15
		bonus_pop_growth_mult = 0.15
		planet_resettlement_unemployed_destination_mult = 0.5
		pop_environment_tolerance = 1.0
	}

	min_distance_from_sun = @habitable_min_distance
	max_distance_from_sun = @habitable_max_distance
	spawn_odds = 0

	extra_orbit_size = 0
	extra_planet_count = 0

	fixed_city_level = 6

	chance_of_ring = 0.2

	planet_size = { min = @habitable_planet_min_size max = @habitable_planet_max_size }
	moon_size = { min = @habitable_moon_min_size max = @habitable_moon_max_size }

	colonizable = yes
	district_set = city_world
	starting_planet = no
	ideal = yes
	uses_alternative_skies_for_moons = no

	carry_cap_per_free_district = @carry_cap_high

	# Per-flavor arcology icons. First match wins; sprites in interface/pd_arcology_variant_icons.gfx.
	conditional_icon = { trigger = { has_modifier = pd_garden_arcology }     icon = GFX_planet_type_pd_garden_arcology     icon_large = GFX_planet_type_pd_garden_arcology_big }
	conditional_icon = { trigger = { has_modifier = pd_fortress_arcology }   icon = GFX_planet_type_pd_fortress_arcology   icon_large = GFX_planet_type_pd_fortress_arcology_big }
	conditional_icon = { trigger = { has_modifier = pd_commercial_arcology } icon = GFX_planet_type_pd_commercial_arcology icon_large = GFX_planet_type_pd_commercial_arcology_big }
}
