# Rare-world specialization zones. They piggyback the vanilla energy/urban zone sets and gate on the world's identity modifier, so they need no district or slot overrides.
# num_zones + planet_limit give the one-of-three choice, and convert_to falls back to the plain vanilla zone if the world loses its modifier. Job count matches the vanilla zone, so specialization pays off through stronger jobs, not more of them.

# Geothermal: channel the generator output into power (grid), alloys (foundry), or research (lab).
zone_pd_geothermal_grid = {
	icon = GFX_district_specialization_energy
	base_buildtime = @zone_buildtime

	potential = {
		hidden_trigger = { exists = owner  owner = { is_wilderness_empire = no } }
		NOT = { has_modifier = resort_colony }
		has_modifier = pd_geothermal
		num_zones = { type = zone_pd_geothermal_foundry value = 0 }
		num_zones = { type = zone_pd_geothermal_lab value = 0 }
	}

	unlock = {
		hidden_trigger = { exists = owner  owner = { is_wilderness_empire = no } }
		custom_tooltip = {
			fail_text = pd_rare_zone_committed_fail
			NOT = { planet = { has_planet_flag = pd_rare_zone_committed } }
		}
	}

	on_queued = { planet = { set_planet_flag = pd_rare_zone_committed } }
	on_unqueued = { planet = { remove_planet_flag = pd_rare_zone_committed } }
	on_built = { planet = { remove_planet_flag = pd_rare_zone_committed } }

	planet_limit = { base = 1 }

	resources = {
		category = planet_zones
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = alloys
			COST = @zone_cost
			NOMADIC_COST_MULT = 0.5
		}
	}

	zone_sets = { energy }

	included_building_sets = {
		generator
		fallen_empire_generator
		generator_automation
	}

	convert_to = {
		zone_energy
		zone_energy_nexus
		zone_energy_hive
	}

	triggered_district_planet_modifier = {
		potential = { exists = owner  owner = { is_gestalt = no } }
		job_pd_geothermal_vent_tapper_add = @scaling_district_1_job
	}

	triggered_district_planet_modifier = {
		potential = { exists = owner  owner = { is_gestalt = yes } }
		job_pd_geothermal_vent_tapper_drone_add = @scaling_district_1_job
	}

	planet_modifier = {
		zone_building_slots_add = 3
	}

	ai_priority = 5
}
