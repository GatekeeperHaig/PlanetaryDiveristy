# On terraform: removes the modifier if needed and updates the planet art
planet_event = {
	id = pdascensionworlds.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		pd_is_flooded_planet = yes
		not = {	has_modifier = pd_aw_shroud_planet }
	}

	immediate = {
		switch = {
			trigger = is_planet_class
			pc_pd_wet_gaia = {
				set_planet_entity = {
					picture = pc_aquatic_hot
					entity = ocean_paradise_planet_01_entity
				}
			}
			pc_pd_wet_superhabitable = {
				set_planet_entity = {
					picture = pc_aquatic_hot
					entity = ocean_paradise_planet_01_entity
				}
			}
			pc_pd_wet_cave = {
				set_planet_entity = {
					picture = pc_pd_aquatic_cave
					entity = aquatic_cave_planet_01_entity
				}
			}
			pc_pd_wet_tidally_locked = {
				set_planet_entity = {
					picture = pc_aquatic
					entity = tidallyaquatic_planet_01_entity
				}
			}
			pc_pd_dry_gaia = {
				remove_modifier = flooded_habitat
			}
			pc_pd_cold_gaia = {
				remove_modifier = flooded_habitat
			}
			pc_pd_dry_superhabitable = {
				remove_modifier = flooded_habitat
			}
			pc_pd_cold_superhabitable = {
				remove_modifier = flooded_habitat
			}
			pc_pd_dry_cave = {
				remove_modifier = flooded_habitat
			}
			pc_pd_cold_cave = {
				remove_modifier = flooded_habitat
			}
			pc_pd_dry_tidally_locked = {
				remove_modifier = flooded_habitat
			}
			pc_pd_cold_tidally_locked = {
				remove_modifier = flooded_habitat
			}
		}
	}
}

# Keeps Flooded Shroud Worlds flooded after terraforming
planet_event = {
	id = pdascensionworlds.150
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_modifier = pd_aw_shroud_planet
		pd_is_flooded_planet = yes
	}

	immediate = {
		IF = {
			limit = {
				pd_is_planet_class_superhabitable = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = pdshroud_planet_01_entity
			}
		}
		IF = {
			limit = {
				pd_is_planet_class_tidally_locked = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = tidallyshroud_planet_01_entity
			}
		}
		IF = {
			limit = {
				pd_is_planet_class_cave = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud_cave
				entity = pdshroud_cave_planet_01_entity
			}
		}
		IF = {
			limit = {
				pd_is_planet_class_pd_gaia = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = pdshroud_planet_01_entity
			}
		}
		# pc_ocean and other flooded vanilla classes use the aquatic shroud entity, like the tidal/super/pd_gaia branches.
		IF = {
			limit = {
				OR = {
					is_planet_class = pc_ocean
					is_planet_class = pc_continental
					is_planet_class = pc_tropical
					is_planet_class = pc_desert
					is_planet_class = pc_arid
					is_planet_class = pc_savannah
					is_planet_class = pc_arctic
					is_planet_class = pc_alpine
					is_planet_class = pc_tundra
					is_planet_class = pc_volcanic
				}
			}
			set_planet_entity = {
				picture = pc_aquatic_shroud_hot
				entity = ocean_paradise_shroud_planet_01_entity
			}
		}
	}
}

# Keeps Shroud Worlds shrouded after terraforming
planet_event = {
	id = pdascensionworlds.200
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_modifier = pd_aw_shroud_planet
		NOT = {
			pd_is_flooded_planet = yes
		}
	}

	immediate = {
		IF = {
			limit = {
				pd_is_planet_class_superhabitable = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = pdshroud_planet_01_entity
			}
		}
		IF = {
			limit = {
				pd_is_planet_class_tidally_locked = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = tidallyshroud_planet_01_entity
			}
		}
		IF = {
			limit = {
				pd_is_planet_class_cave = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud_cave
				entity = pdshroud_cave_planet_01_entity
			}
		}
		IF = {
			limit = {
				pd_is_planet_class_pd_gaia = yes
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = pdshroud_planet_01_entity
			}
		}
		# Volcanic has the only class-specific vanilla shroud art; .200 runs after pdengine.160 in pd_on_actions, so it re-overrides .160's PD visual back to shroud.
		IF = {
			limit = {
				is_planet_class = pc_volcanic
			}
			set_planet_entity = {
				picture = pc_pd_volcanic_shroud
				entity = pd_volcanic_shroud_planet_01_entity
			}
		}
		#Other vanilla habitable classes fall back to the generic shroud entity.
		IF = {
			limit = {
				OR = {
					is_planet_class = pc_continental
					is_planet_class = pc_tropical
					is_planet_class = pc_desert
					is_planet_class = pc_arid
					is_planet_class = pc_savannah
					is_planet_class = pc_arctic
					is_planet_class = pc_alpine
					is_planet_class = pc_tundra
				}
			}
			set_planet_entity = {
				picture = pc_pd_shroud
				entity = pdshroud_planet_01_entity
			}
		}
	}
}

#This event fires to give Necro Empires their Uncommon Tomb Habitability
country_event = {
	id = pdascensionworlds.300
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_ascension_perk = ap_pd_necro_worlds
		NOT = {
			has_country_flag = ap_pd_necro_worlds_uncommon_modifiers
		}
	}

	immediate = {
		give_technology = {
			tech = tech_pd_uncommon_necro_habitability
			message = no
		}
		set_country_flag = ap_pd_necro_worlds_uncommon_modifiers
	}
}
