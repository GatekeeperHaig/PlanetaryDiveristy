# Whitens every portrait of a Biosynthetic species (pops, leaders, species view), reusing the masking_texture vanilla uses for holographic.
# Rides Machine Age's holographic engine hook and gates on the class trait, so no flag plumbing is needed.
pd_biosynth_glow = {
	trigger = {
		OR = {
			is_scope_type = leader
			is_scope_type = pop
			is_scope_type = pop_group
			is_scope_type = species
		}
		species = {
			has_trait = trait_biosynth
		}
	}

	masking_texture = "gfx/interface/leaders/portrait_mask_fade.dds"
	effectFile = "gfx/FX/buttonstate_rendertarget_pd_biosynth.shader"
}
