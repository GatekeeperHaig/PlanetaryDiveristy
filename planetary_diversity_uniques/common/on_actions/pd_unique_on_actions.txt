# A ship has surveyed a planet.
# Scope = Ship
# From = Planet
on_survey_planet = {
	events = {
		pdfloating.100
		pdcrystal.100
		pdbiosynth.100
		pdbonefield.20
	}
}

# A ship has entered a system.
# Scope = Ship
# From = System
on_entering_system = {
	events = {
		pdbonefield.10
	}
}

on_planet_class_changed = {
	events = {
		pdunique.200
		pdfloating.200
		pdfloating.210
	}
}
