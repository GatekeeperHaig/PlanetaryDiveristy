# Overrides vanilla decision_arcology_project_relic to auto-install zone_pd_archeotech_preserve into the first buildable zone slot.
# One add_zone call is appended to the vanilla effect block.
decision_arcology_project_relic = {
	owned_planets_only = yes
	icon = decision_arcology_project

	enactment_time = {
		base = 3600
		modifier = {
			owner = { is_megacity_architects_empire = yes }
			factor = 0.5
		}
	}
	resources = {
		category = decisions
		cost = {
			influence = 200
			alloys = 10000
			mult = owner.value:scripted_modifier_mult|MODIFIER|arcology_project_cost_mult|
		}
	}

	potential = {
		is_planet_class = pc_relic
		is_artificial = no
		exists = owner
		owner = {
			OR = {
				is_regular_empire = yes
				has_valid_civic = civic_machine_servitor
			}
		}
	}

	allow = {
		num_uncleared_blockers < 1
		owner = { has_technology = tech_housing_2 }
		if = {
			limit = { owner = { is_megacity_architects_empire = yes } }
			custom_tooltip = {
				fail_text = "needs_urban_expansion_frame"
				has_modifier = urban_expansion_frame
			}
		}
	}

	on_queued = {
		set_carrier_flag = arcology_project_construction
	}

	on_unqueued = {
		remove_carrier_flag = arcology_project_construction
	}

	effect = {
		remove_carrier_flag = arcology_project_construction
		# Make city-planet
		custom_tooltip = decision_arcology_project_effects
		hidden_effect = {
			save_deposits = yes
			clear_deposits = yes
			restore_saved_deposits = yes
			change_pc = pc_city
			relic_world_conversion_effect = {
				DISTRICT = district_arcology_housing
				VARIABLE = num_housing_districts
			}
			set_planet_entity = {
				entity = "city_planet_entity"
				graphical_culture = owner
			}
			if = {
				limit = { has_pd_capital_deposit = yes }
				pd_resync_capital_art_effect = yes
			}
			carrier_event = { id = action.68 }
			add_deposit = d_former_relic_world
			carrier_event = { id = mega.201 }
			while = {
				count = 4
				solar_system = {
					random_system_ambient_object = {
						limit = { has_ambient_object_flag = relic_world_debris }
						destroy_ambient_object = this
					}
				}
			}
			if = {
				limit = {
					OR = {
						is_active_resolution = "resolution_ecology_environmental_control_board"
						is_active_resolution = "resolution_ecology_paradise_initiative"
					}
				}
				owner = {
					set_timed_country_flag = { flag = resolution_breached_terraformed days = 3600 }
				}
			}
		}
		add_zone = {
			district = 0
			zone = zone_pd_archeotech_preserve
			zone_slot = 1
			replace = yes
		}
	}

	ai_weight = {
		weight = 100
	}
}
