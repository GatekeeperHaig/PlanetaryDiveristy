# Custom jobs for the rare-world specialization zones. Each is a stronger version of the vanilla job it stands in for, so the zone keeps the vanilla job count but out-produces a normal zone.
# resources.category points at the vanilla category so the usual production/upkeep modifiers still apply. Gestalt drone variants split off only for the tier swap (worker->simple_drone, specialist->complex_drone); hive and machine share one drone since their output is identical.

# Geothermal grid: Vent Tapper (energy). Vanilla technician makes energy 6.
pd_geothermal_vent_tapper = {
	category = worker

	swappable_data = {
		default = {
			condition_string = WORKER_JOB_TRIGGER
			building_icon = building_energy_grid
			icon = technician
		}
	}

	tags = { energy }

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_worker_job
	possible = {
		hidden_trigger = { exists = owner }
		owner = { is_gestalt = no }
	}

	resources = {
		category = planet_technician
		produces = {
			energy = 8
			minerals = 1
		}
	}

	weight = {
		weight = @worker_job_weight
	}
}
